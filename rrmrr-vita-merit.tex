% Narrower label column than rrmrr-cv.sty (18% instead of 26%), so entries
% get more of the line. Otherwise the same as \cvsection in the style.
\renewcommand{\cvsection}[2]{%
  \par\vspace{\cvgap}
  \begin{list}{}{%
      \setlength{\leftmargin}{0.18\textwidth}
      \setlength{\labelwidth}{0pt}
      \setlength{\labelsep}{0pt}
      \setlength{\topsep}{0pt}
      \setlength{\partopsep}{0pt}
      \setlength{\itemsep}{0pt}
      \setlength{\parsep}{0pt}
    }
  \item
    \makebox[0pt][r]{%
      \smash{%
        \begin{minipage}[t]{0.165\textwidth}
          \raggedright\textbf{#1}
        \end{minipage}%
      }%
      \hspace{0.015\textwidth}%
    }%
    \ignorespaces
    #2
  \end{list}
}

% Talks: a narrower year column than the style's (years only here).
\renewcommand{\cvitemconf}[2]{%
  \noindent
  \begin{minipage}[t]{0.91\linewidth}
    \raggedright \textnormal{#1}\strut
  \end{minipage}%
  \hfill%
  \begin{minipage}[t]{0.09\linewidth}
    \raggedleft {\small #2}\strut
  \end{minipage}%
  \par
  \nointerlineskip
  \vspace{\cvgap}%
}

% Same as \cvpub in the style, but the title takes all of the line the
% year leaves free (the full line when there is no year).
\newlength\cvyearwidth
\renewcommand{\cvpub}[5][]{%
  \noindent
  \ifblank{#5}{%
    \begin{minipage}[t]{\linewidth}
      \raggedright ``\ifblank{#1}{#2}{\href{#1}{#2}}''\strut
    \end{minipage}%
  }{%
    \settowidth\cvyearwidth{\small #5}%
    \begin{minipage}[t]{\dimexpr\linewidth-\cvyearwidth-1.5em\relax}
      \raggedright ``\ifblank{#1}{#2}{\href{#1}{#2}}''\strut
    \end{minipage}%
    \hfill%
    {\small #5}\strut
  }%
  \par\nopagebreak\nointerlineskip\vspace{\cvsubgap}%
  \ifboolexpr{test {\ifblank{#3}} and test {\ifblank{#4}}}{}{%
    {%
      \raggedright
      \addtolength{\leftskip}{1em}%
      \small
      \noindent #3\ifboolexpr{test {\ifblank{#3}} or test {\ifblank{#4}}}{}{. }#4\par
    }%
  }%
  \vspace{\cvgap}%
}
