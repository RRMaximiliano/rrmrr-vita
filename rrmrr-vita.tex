% --- 1. Main Section Container (\cvsection) ---
% Creates a two-column layout:
% Left: Section Title (e.g., "Education")
% Right: Content List (allows page breaks)
% ------------------------------------------------------------------------
\newcommand{\cvsection}[2]{%
  \par\vspace{4pt}
  \begin{list}{}{%
      \setlength{\leftmargin}{0.26\textwidth}
      \setlength{\labelwidth}{0pt}
      \setlength{\labelsep}{0pt}
      \setlength{\topsep}{0pt}
      \setlength{\partopsep}{0pt}
      \setlength{\itemsep}{0pt}
      \setlength{\parsep}{0pt}
    }
  \item
    \makebox[0pt][r]{%
      % \smash hides the vertical height/depth of the title.
      % This prevents long titles from messing up the first item's spacing.
      \smash{%
        \begin{minipage}[t]{0.24\textwidth}
          \raggedright\textbf{#1}
        \end{minipage}%
      }%
      \hspace{0.02\textwidth}%
    }%
    \ignorespaces
    #2
  \end{list}
}


\newif\ifcvfirstsub % set by \cvitem, read by \cvsubitem

% --- 2. Standard Item (\cvitem) ---
% Used for: Education, Teaching Experience, Awards
% Args: [Font Style] {Title} {Description} {Date}
% Titles are regular weight; bold is reserved for section labels and for
% my name in author lists. Description lines take no final period.
% ------------------------------------------------------------------------
\newcommand{\cvitem}[4][\textnormal]{%
  \noindent
  {#1{#2}}\strut \hfill {\small #4}% <--- Year is now forced to SMALL
  \par\nopagebreak % keep the title with its description

  \ifx&#3&%
  % Case: No Description
  \vspace{4pt}
  \else
  % Case: Has Description
  \nointerlineskip
  \vspace{2pt}
  {%
    \raggedright % ragged like \cvpub; resets \leftskip, so indent after
    \addtolength{\leftskip}{1em}
    \small % <--- Forces the entire Description block to SMALL
    \cvfirstsubtrue
    #3% no \noindent: it would leave an empty line before a \cvsubitem
    \par
  }%
  \vspace{4pt}
  \fi
}


% --- 3. Conference/Seminar Item (\cvitemconf) ---
% Used for: Lists of talks where text might wrap
% Uses columns so the Date stays top-aligned even if text wraps
% ------------------------------------------------------------------------
\newcommand{\cvitemconf}[2]{%
  \noindent
  % Left Column
  \begin{minipage}[t]{0.87\linewidth}
    \raggedright \textnormal{#1}\strut % \strut ensures consistent line height
  \end{minipage}%
  \hfill%
  % Right Column
  \begin{minipage}[t]{0.13\linewidth}
    \raggedleft {\small #2}\strut
  \end{minipage}%
  \par
  % THE FIX:
  \nointerlineskip % 1. Ignore variable baseline calculations
  \vspace{4pt}     % 2. Add consistent spacing (Ink-to-Ink)
}

% --- 4. Sub-entry (\cvsubitem) ---
% Used for: courses under one teaching role
% Args: {Title} {Detail, e.g. instructors (italic; may be empty)} {Date}
% ------------------------------------------------------------------------
\newcommand{\cvsubitem}[3]{%
  \par
  \ifcvfirstsub\cvfirstsubfalse\else\addvspace{2pt}\fi % gap between, not above
  \noindent #1\hfill #3\par
  \ifx&#2&\else
    \nopagebreak\noindent\textit{#2}\par
  \fi
}

% --- 5. Publication (\cvpub) ---
% Used for: publications, working papers, and work in progress
% Args: {Title} {Authors (may be empty)} {Venue (may be empty)} {Year}
% Title on its own line with the year on the right; authors and venue
% on one line below in small type, so every entry reads the same way.
% ------------------------------------------------------------------------
\newcommand{\cvpub}[4]{%
  \noindent
  \begin{minipage}[t]{0.83\linewidth}
    \raggedright #1\strut
  \end{minipage}%
  \hfill%
  \begin{minipage}[t]{0.16\linewidth}
    \raggedleft {\small #4}\strut
  \end{minipage}%
  \par\nopagebreak\nointerlineskip\vspace{2pt}
  \ifx&#2#3&\else
    {%
      \raggedright
      \addtolength{\leftskip}{1em}% before \small: same indent as \cvitem
      \small
      \noindent #2\ifx&#2&\else\ifx&#3&\else. \fi\fi #3\par
    }%
  \fi
  \vspace{4pt}
}

% --- 6. Sub-heading (\cvsubheading) ---
% Used for: groups inside a section (e.g. Journal Articles)
% ------------------------------------------------------------------------
\newcommand{\cvsubheading}[1]{%
  \ifvmode\addvspace{4pt}\fi % no gap when it opens the section
  \noindent\textit{#1}\par\nopagebreak\vspace{3pt}
}
